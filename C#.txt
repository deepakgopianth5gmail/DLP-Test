using Google.Cloud.Dlp.V2;
using Google.Protobuf.WellKnownTypes;
using System;

class DlpSample
{
    static void Main(string[] args)
    {
        // Your Google Cloud project ID
        string projectId = "[PROJECT_ID]";

        // Initialize the DLP client
        DlpServiceClient dlpClient = DlpServiceClient.Create();

        // Define the item to inspect
        ContentItem item = new ContentItem
        {
            Value = "My email is example@example.com."
        };

        // Set the info types you want to detect (e.g., EMAIL_ADDRESS)
        InspectConfig inspectConfig = new InspectConfig
        {
            InfoTypes = { new InfoType { Name = "EMAIL_ADDRESS" } }
        };

        // Construct the request parent path
        ProjectName projectName = new ProjectName(projectId);

        // Call the DLP API to inspect the content
        var response = dlpClient.InspectContent(projectName, inspectConfig, item);

        // Check and print findings
        if (response.Result.Findings.Count > 0)
        {
            Console.WriteLine("Findings:");
            foreach (var finding in response.Result.Findings)
            {
                Console.WriteLine($"  InfoType: {finding.InfoType.Name}");
                Console.WriteLine($"  Quote: {finding.Quote}");
                Console.WriteLine($"  Likelihood: {finding.Likelihood}");
                Console.WriteLine();
            }
        }
        else
        {
            Console.WriteLine("No findings.");
        }
    }
}
