
def inspect_sensitive_data(project_id, text_to_inspect):
    # Initialize the DLP client
    client = dlp_v2.DlpServiceClient()

    # Specify the project to associate with the request
    parent = f"projects/{project_id}"

    # Configure the info types you want to detect - here, EMAIL_ADDRESS
    info_types = [{"name": "EMAIL_ADDRESS"}]

    # Create the inspect config specifying which info types to look for
    inspect_config = {"info_types": info_types}

    # Provide the content to be inspected
    item = {"value": text_to_inspect}

    # Construct the inspect request
    response = client.inspect_content(
        request={
            "parent": parent,
            "inspect_config": inspect_config,
            "item": item,
        }
    )
    
    # Parse and print the findings
    if response.result.findings:
        print("Findings:")
        for finding in response.result.findings:
            print(f" - Quote: {finding.quote}")
            print(f" - Info type: {finding.info_type.name}")
            print(f" - Likelihood: {finding.likelihood.name}")
    else:
        print("No findings.")

# Example usage:
if __name__ == "__main__":
    project_id = "your-google-cloud-project-id"  # Replace with your project ID
    text = "My email is example@example.com."
    inspect_sensitive_data(project_id, text)
